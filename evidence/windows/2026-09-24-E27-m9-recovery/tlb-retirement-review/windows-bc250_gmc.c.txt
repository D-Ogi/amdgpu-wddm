152: int bc250_gmc_flush_gpu_tlb(struct amdgpu_device *adev, u32 vmid, u32 vmhub, u32 flush_type)
153: {
154: 	struct amdgpu_vmhub *hub;
155: 	bool use_semaphore;
156: 	bool semaphore_taken = false;
157: 	u32 inv_req, sem, req, ack;
158: 	unsigned int i;
159: 	int r = 0;
160: 
161: 	if (adev == NULL || vmhub >= AMDGPU_MAX_VMHUBS)
162: 		return BC250_EINVAL;
163: 	hub = &adev->vmhub[vmhub];
164: 	if (hub->vmhub_funcs == NULL)
165: 		return BC250_EINVAL;    /* the hub's init() has not run */
166: 
167: 	use_semaphore = (vmhub == AMDGPU_MMHUB0(0));
168: 	inv_req = hub->vmhub_funcs->get_invalidate_req(vmid, flush_type);
169: 	sem = hub->vm_inv_eng0_sem + hub->eng_distance * BC250_GART_ENGINE;
170: 	req = hub->vm_inv_eng0_req + hub->eng_distance * BC250_GART_ENGINE;
171: 	ack = hub->vm_inv_eng0_ack + hub->eng_distance * BC250_GART_ENGINE;
172: 
173: 	/* "It may lose gpuvm invalidate acknowledge state across power-gating off cycle": take the
174: 	 * semaphore before invalidating and release it afterwards. A read returning 1 is an
175: 	 * acquire - which is also why reading these registers is forbidden outside this path
176: 	 * (facts M25). */
177: 	if (use_semaphore) {
178: 		for (i = 0; i < adev->usec_timeout; i++) {
179: 			if (RREG32(sem) & 0x1) {
180: 				semaphore_taken = true;
181: 				break;
182: 			}
183: 			bc250_shim_udelay(1);
184: 		}
185: 		if (!semaphore_taken) {
186: 			/* Upstream logs and carries on; the invalidation below still has to happen,
187: 			 * and the semaphore still has to be written back to 0 afterwards, because a
188: 			 * read may have acquired it even though none returned 1. */
189: 			dev_err(adev->dev, "Timeout waiting for sem acquire in VM flush!\n");
190: 			r = BC250_ETIME;
191: 		}
192: 	}
193: 
194: 	WREG32(req, inv_req);
195: 
196: 	if (vmhub == AMDGPU_GFXHUB(0) &&
197: 	    amdgpu_ip_version(adev, GC_HWIP, 0) < IP_VERSION(10, 3, 0))
198: 		RREG32(req);
199: 
200: 	for (i = 0; i < adev->usec_timeout; i++) {
201: 		if (RREG32(ack) & (1u << vmid))
202: 			break;
203: 		bc250_shim_udelay(1);
204: 	}
205: 	if (i >= adev->usec_timeout) {
206: 		dev_err(adev->dev, "Timeout waiting for VM flush hub: %d!\n", (int)vmhub);
207: 		if (r == 0)
208: 			r = BC250_ETIME;
209: 	}
210: 
211: 	/* Released on every path that reached the invalidation, the acknowledge timeout included:
212: 	 * upstream releases it unconditionally, and leaving it held would wedge every later flush. */
213: 	if (use_semaphore)
214: 		WREG32(sem, 0);
215: 
216: 	return r;
217: }
271: int bc250_gmc_gart_enable(struct amdgpu_device *adev)
272: {
273: 	int r;
274: 
275: 	if (adev == NULL || adev->gart.bo == NULL)
276: 		return BC250_EINVAL;
277: 
278: 	r = adev->gfxhub.funcs->gart_enable(adev);
279: 	if (r)
280: 		return r;
281: 
282: 	r = adev->mmhub.funcs->gart_enable(adev);
283: 	if (r)
284: 		return r;
285: 
286: 	/* Upstream: adev->hdp.funcs->init_registers(adev), then amdgpu_device_flush_hdp(adev, NULL).
287: 	 * HDP is not imported yet. Its only write here is HDP_MISC_CNTL, and the flush is a no-op on
288: 	 * an APU. */
289: 
290: 	/* value = amdgpu_vm_fault_stop != AMDGPU_VM_FAULT_STOP_ALWAYS; the module parameter defaults
291: 	 * to AMDGPU_VM_FAULT_STOP_NEVER, so value is true: faults go to the default page instead of
292: 	 * stopping the engine. */
293: 	adev->gfxhub.funcs->set_fault_enable_default(adev, true);
294: 	adev->mmhub.funcs->set_fault_enable_default(adev, true);
295: 
296: 	/* Upstream calls these two for their side effect and ignores the result; a flush timeout is
297: 	 * not a reason to fail the bring-up, and on unit A amdgpu did time out here (the pre-driver
298: 	 * sweep had taken the MMHUB semaphore, facts M25) and went on to a working GPU. The code is
299: 	 * logged by bc250_gmc_flush_gpu_tlb itself; a caller that wants it calls that directly. */
300: 	(void)bc250_gmc_flush_gpu_tlb(adev, 0, AMDGPU_MMHUB0(0), 0);
301: 	(void)bc250_gmc_flush_gpu_tlb(adev, 0, AMDGPU_GFXHUB(0), 0);
302: 
303: 	return 0;
304: }
